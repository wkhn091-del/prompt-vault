Agency Starter Kit — Next.js 14 + Sanity + Tailwind + Framer Motion
מדריך מלא להקמת תיקיית בסיס לפרויקטים חדשים.


________________


שלב 1: פקודות CLI
npx create-next-app@latest my-agency-starter --typescript --tailwind --app --src-dir=false --import-alias "@/*"


cd my-agency-starter


npm install sanity @sanity/vision next-sanity @sanity/image-url styled-components


npm install framer-motion

sanity/schemas/seo.ts
import { defineField, defineType } from 'sanity'


export default defineType({


  name: 'seo',


  title: 'SEO',


  type: 'object',


  fields: [


    defineField({


      name: 'metaTitle',


      title: 'כותרת (Meta Title)',


      type: 'string',


      validation: (Rule) => Rule.max(60).warning('מומלץ עד 60 תווים'),


    }),


    defineField({


      name: 'metaDescription',


      title: 'תיאור (Meta Description)',


      type: 'text',


      rows: 3,


      validation: (Rule) => Rule.max(160).warning('מומלץ עד 160 תווים'),


    }),


    defineField({


      name: 'shareImage',


      title: 'תמונת שיתוף (OG Image)',


      type: 'image',


      options: { hotspot: true },


    }),


    defineField({


      name: 'noIndex',


      title: 'הסתר ממנועי חיפוש',


      type: 'boolean',


      initialValue: false,


    }),


  ],


})

          type: 'block',


        },


      ],


    }),

    }),


  ],
